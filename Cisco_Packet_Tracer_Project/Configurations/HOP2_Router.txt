! HOP2 Router Configuration
! Intermediate node between HOP1 and Mumbai
! Bandwidth: 20 Mbps, Cost: 10

enable
configure terminal

! Basic router configuration
hostname HOP2
enable secret cisco123
service password-encryption

! Interface configurations
interface FastEthernet0/0
 description Connection to HOP1
 ip address 10.0.3.2 255.255.255.0
 ip ospf cost 10
 bandwidth 20000
 no shutdown

interface FastEthernet0/1
 description Connection to Mumbai
 ip address 10.0.2.1 255.255.255.0
 ip ospf cost 10
 bandwidth 100000
 no shutdown

interface Loopback0
 description HOP2 Loopback
 ip address 10.10.2.1 255.255.255.0

! OSPF Configuration
router ospf 1
 router-id 10.10.2.1
 network 10.0.2.0 0.0.0.255 area 0
 network 10.0.3.0 0.0.0.255 area 0
 network 10.10.2.0 0.0.0.255 area 0
 passive-interface default
 no passive-interface FastEthernet0/0
 no passive-interface FastEthernet0/1

! RIP Configuration (Alternative)
router rip
 version 2
 network 10.0.2.0
 network 10.0.3.0
 network 10.10.2.0
 no auto-summary

! Static routes (backup)
ip route 0.0.0.0 0.0.0.0 10.0.2.2
ip route 192.168.2.0 255.255.255.0 10.0.3.1

! Access control
access-list 1 permit 10.0.2.0 0.0.0.255
access-list 1 permit 10.0.3.0 0.0.0.255
access-list 1 permit 10.10.2.0 0.0.0.255

! Logging
logging buffered 4096
logging console

! Save configuration
write memory
end
