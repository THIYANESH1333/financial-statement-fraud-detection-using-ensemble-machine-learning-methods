! HOP6 Router Configuration
! Final intermediate node between HOP5 and Chennai
! Bandwidth: 170 Mbps, Cost: 10

enable
configure terminal

! Basic router configuration
hostname HOP6
enable secret cisco123
service password-encryption

! Interface configurations
interface FastEthernet0/0
 description Connection to HOP5
 ip address 10.1.5.2 255.255.255.0
 ip ospf cost 10
 bandwidth 100000
 no shutdown

interface FastEthernet0/1
 description Connection to Chennai
 ip address 10.1.4.1 255.255.255.0
 ip ospf cost 10
 bandwidth 170000
 no shutdown

interface Loopback0
 description HOP6 Loopback
 ip address 10.10.6.1 255.255.255.0

! OSPF Configuration
router ospf 1
 router-id 10.10.6.1
 network 10.1.4.0 0.0.0.255 area 0
 network 10.1.5.0 0.0.0.255 area 0
 network 10.10.6.0 0.0.0.255 area 0
 passive-interface default
 no passive-interface FastEthernet0/0
 no passive-interface FastEthernet0/1

! RIP Configuration (Alternative)
router rip
 version 2
 network 10.1.4.0
 network 10.1.5.0
 network 10.10.6.0
 no auto-summary

! Static routes (backup)
ip route 0.0.0.0 0.0.0.0 10.1.4.2
ip route 192.168.1.0 255.255.255.0 10.1.5.1

! Access control
access-list 1 permit 10.1.4.0 0.0.0.255
access-list 1 permit 10.1.5.0 0.0.0.255
access-list 1 permit 10.10.6.0 0.0.0.255

! Logging
logging buffered 4096
logging console

! Save configuration
write memory
end
