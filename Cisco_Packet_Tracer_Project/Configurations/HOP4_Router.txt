! HOP4 Router Configuration
! Intermediate node between HOP3 and HOP5
! Cost: 20

enable
configure terminal

! Basic router configuration
hostname HOP4
enable secret cisco123
service password-encryption

! Interface configurations
interface FastEthernet0/0
 description Connection to HOP3
 ip address 10.1.2.2 255.255.255.0
 ip ospf cost 20
 bandwidth 100000
 no shutdown

interface FastEthernet0/1
 description Connection to HOP5
 ip address 10.1.3.1 255.255.255.0
 ip ospf cost 20
 bandwidth 100000
 no shutdown

interface Loopback0
 description HOP4 Loopback
 ip address 10.10.4.1 255.255.255.0

! OSPF Configuration
router ospf 1
 router-id 10.10.4.1
 network 10.1.2.0 0.0.0.255 area 0
 network 10.1.3.0 0.0.0.255 area 0
 network 10.10.4.0 0.0.0.255 area 0
 passive-interface default
 no passive-interface FastEthernet0/0
 no passive-interface FastEthernet0/1

! RIP Configuration (Alternative)
router rip
 version 2
 network 10.1.2.0
 network 10.1.3.0
 network 10.10.4.0
 no auto-summary

! Static routes (backup)
ip route 0.0.0.0 0.0.0.0 10.1.2.1
ip route 192.168.2.0 255.255.255.0 10.1.3.2

! Access control
access-list 1 permit 10.1.2.0 0.0.0.255
access-list 1 permit 10.1.3.0 0.0.0.255
access-list 1 permit 10.10.4.0 0.0.0.255

! Logging
logging buffered 4096
logging console

! Save configuration
write memory
end
